cmake_minimum_required(VERSION 3.24)
project(t11_ffn LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(t11-ffn ../../models/deepseek41/ffn.hip ffn_fixture.cpp ../../models/deepseek41/artifact.cpp ../../native/quant/quant.hip ../../native/artifact/native_reader.cpp ../../native/stats/measurement.cpp)
target_include_directories(t11-ffn PRIVATE ../../models/deepseek41 ../../native/artifact "${ROOT}/include")
target_compile_features(t11-ffn PRIVATE cxx_std_17)
set_target_properties(t11-ffn PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_options(t11-ffn PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t11-ffn PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T11_SOURCE_ID="${T11_SOURCE_ID}")
target_link_libraries(t11-ffn PRIVATE hip::host psapi)
