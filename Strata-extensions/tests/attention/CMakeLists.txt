cmake_minimum_required(VERSION 3.24)
project(t15_attention LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(t15-attention ../../models/deepseek41/attention.hip attention_fixture.cpp ../../models/deepseek41/kv.hip ../../models/deepseek41/artifact.cpp ../../native/quant/quant.hip ../../native/artifact/native_reader.cpp ../../native/stats/measurement.cpp)
target_include_directories(t15-attention PRIVATE ../../models/deepseek41 ../../native/artifact "${ROOT}/include")
target_compile_features(t15-attention PRIVATE cxx_std_17)
set_target_properties(t15-attention PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_options(t15-attention PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t15-attention PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T15_SOURCE_ID="${T15_SOURCE_ID}")
target_link_libraries(t15-attention PRIVATE hip::host psapi)
