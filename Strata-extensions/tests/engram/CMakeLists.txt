cmake_minimum_required(VERSION 3.24)
project(t13_engram LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MATERIAL "")
foreach(INPUT Strata-extensions/models/deepseek41/engram.hpp Strata-extensions/models/deepseek41/engram.cpp Strata-extensions/models/deepseek41/engram.hip Strata-extensions/tests/engram/engram_fixture.cpp)
  file(SHA256 "${ROOT}/${INPUT}" HASH)
  string(APPEND MATERIAL "${INPUT}|${HASH}\n")
endforeach()
string(SHA256 SOURCE_ID "${MATERIAL}")
add_executable(t13-engram engram_fixture.cpp ../../models/deepseek41/engram.cpp ../../models/deepseek41/engram.hip ../../models/deepseek41/artifact.cpp ../../native/artifact/native_reader.cpp ../../native/io/span_provider.cpp ../../native/cache/lease_queue.cpp ../../native/stats/measurement.cpp "${ROOT}/src/platform/direct_file.cpp")
target_include_directories(t13-engram PRIVATE ../../models/deepseek41 ../../native/artifact ../../native/io ../../native/cache ../../native/stats "${ROOT}/include")
target_compile_features(t13-engram PRIVATE cxx_std_17)
set_target_properties(t13-engram PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_options(t13-engram PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t13-engram PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T13_SOURCE_ID="${SOURCE_ID}")
target_link_libraries(t13-engram PRIVATE hip::host psapi)
