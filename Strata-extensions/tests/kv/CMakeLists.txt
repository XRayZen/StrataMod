cmake_minimum_required(VERSION 3.24)
project(t14_kv LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(t14-kv ../../models/deepseek41/kv.hip kv_fixture.cpp ../../models/deepseek41/artifact.cpp ../../native/quant/quant.hip ../../native/artifact/native_reader.cpp ../../native/stats/measurement.cpp)
target_include_directories(t14-kv PRIVATE ../../models/deepseek41 ../../native/artifact "${ROOT}/include")
target_compile_features(t14-kv PRIVATE cxx_std_17)
set_target_properties(t14-kv PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_options(t14-kv PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t14-kv PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T14_SOURCE_ID="${T14_SOURCE_ID}")
target_link_libraries(t14-kv PRIVATE hip::host psapi)
