cmake_minimum_required(VERSION 3.24)
project(t12_mhc LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(t12-mhc ../../models/deepseek41/mhc.hip mhc_fixture.cpp ../../models/deepseek41/artifact.cpp ../../native/quant/quant.hip ../../native/artifact/native_reader.cpp ../../native/stats/measurement.cpp)
target_include_directories(t12-mhc PRIVATE ../../models/deepseek41 ../../native/artifact "${ROOT}/include")
target_compile_features(t12-mhc PRIVATE cxx_std_17)
set_target_properties(t12-mhc PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_options(t12-mhc PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t12-mhc PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T12_SOURCE_ID="${T12_SOURCE_ID}")
target_link_libraries(t12-mhc PRIVATE hip::host psapi)
