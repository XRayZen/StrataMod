cmake_minimum_required(VERSION 3.24)
project(t18_supply LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MATERIAL "")
foreach(INPUT Strata-extensions/native/quant/quant.hpp Strata-extensions/native/quant/quant.hip Strata-extensions/native/cache/lease_queue.hpp Strata-extensions/native/cache/lease_queue.cpp Strata-extensions/native/cache/primary_supply.hpp Strata-extensions/native/cache/primary_supply.hip Strata-extensions/tests/primary_supply/fixture.hip)
 file(SHA256 "${ROOT}/${INPUT}" HASH)
 string(APPEND MATERIAL "${INPUT}|${HASH}\n")
endforeach()
string(SHA256 SOURCE_ID "${MATERIAL}")
set_source_files_properties("${ROOT}/src/core/device.cu" PROPERTIES LANGUAGE HIP)
add_executable(t18-supply fixture.hip ../../native/quant/quant.hip ../../native/cache/primary_supply.hip ../../native/cache/lease_queue.cpp ../../native/stats/measurement.cpp ../../native/io/span_provider.cpp ../../native/artifact/native_reader.cpp ../../models/deepseek41/artifact.cpp "${ROOT}/src/platform/direct_file.cpp" "${ROOT}/src/core/device.cu")
target_include_directories(t18-supply PRIVATE ../../native/cache ../../native/stats ../../native/io ../../native/artifact "${ROOT}/include/strata/hip_compat" "${ROOT}/include")
target_compile_features(t18-supply PRIVATE cxx_std_17)
target_compile_options(t18-supply PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t18-supply PRIVATE STRATA_USE_HIP STRATA_HIP_ARCHS="gfx1100,gfx1201" NOMINMAX WIN32_LEAN_AND_MEAN T18_SOURCE_ID="${SOURCE_ID}")
target_link_libraries(t18-supply PRIVATE hip::host psapi)
set_target_properties(t18-supply PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
