cmake_minimum_required(VERSION 3.24)
project(t21_device LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MATERIAL "")
foreach(INPUT Strata-extensions/native/device/roles.hpp Strata-extensions/native/device/roles.hip Strata-extensions/tests/device/fixture.hip)
 file(SHA256 "${ROOT}/${INPUT}" HASH)
 string(APPEND MATERIAL "${INPUT}|${HASH}\n")
endforeach()
string(SHA256 SOURCE_ID "${MATERIAL}")
set_source_files_properties("${ROOT}/src/core/device.cu" PROPERTIES LANGUAGE HIP)
add_executable(t21-device fixture.hip ../../native/device/roles.hip ../../native/cache/lease_queue.cpp ../../native/stats/measurement.cpp "${ROOT}/src/core/device.cu")
target_include_directories(t21-device PRIVATE ../../native/device ../../native/cache ../../native/stats "${ROOT}/include/strata/hip_compat" "${ROOT}/include")
target_compile_features(t21-device PRIVATE cxx_std_17)
target_compile_definitions(t21-device PRIVATE STRATA_USE_HIP STRATA_HIP_ARCHS="gfx1100,gfx1201" NOMINMAX WIN32_LEAN_AND_MEAN T21_SOURCE_ID="${SOURCE_ID}")
target_link_libraries(t21-device PRIVATE hip::host psapi)
set_target_properties(t21-device PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
