# ABI is nonrecursive: source hashes and consumed configured inputs precede ABI define.
get_target_property(T39_HIP_DEFS hip::host INTERFACE_COMPILE_DEFINITIONS)
get_target_property(T39_HIP_OPTIONS hip::host INTERFACE_COMPILE_OPTIONS)
foreach(T39_TARGET Strata-deepseek t28-construction t38-seed-fixture t38-proposal-fixture t38-actor-fixture)
 if(TARGET ${T39_TARGET})
  target_compile_definitions(${T39_TARGET} PRIVATE STRATA_T39_NORMAL_OBSERVER=${T39_OBSERVER_VALUE})
  get_target_property(T39_DEFS ${T39_TARGET} COMPILE_DEFINITIONS)
  set(T39_SOURCE_ID "")
  foreach(T39_DEF ${T39_DEFS})
   if(T39_DEF MATCHES "^T28_SOURCE_ID=\"([^\"]+)\"$")
    set(T39_SOURCE_ID "${CMAKE_MATCH_1}")
   endif()
  endforeach()
  if(NOT T39_SOURCE_ID)
   message(FATAL_ERROR "Graph ABI requires actual target source identity")
  endif()
  list(FILTER T39_DEFS EXCLUDE REGEX "^T28_SOURCE_ID=")
  get_target_property(T39_OPTIONS ${T39_TARGET} COMPILE_OPTIONS)
  set(T39_BODY_DEFS "not-applicable")
  set(T39_BODY_OPTIONS "not-applicable")
  set(T39_BODY "")
  if(T39_TARGET STREQUAL "t38-seed-fixture")
   set(T39_BODY t38_seed_body)
  elseif(T39_TARGET STREQUAL "t38-proposal-fixture")
   set(T39_BODY t38_proposal_body)
  elseif(T39_TARGET STREQUAL "t38-actor-fixture")
   set(T39_BODY t38_actor_body)
  endif()
  if(T39_BODY)
   target_compile_definitions(${T39_BODY} PRIVATE STRATA_T39_NORMAL_OBSERVER=${T39_OBSERVER_VALUE})
   get_target_property(T39_BODY_DEFS ${T39_BODY} COMPILE_DEFINITIONS)
   get_target_property(T39_BODY_OPTIONS ${T39_BODY} COMPILE_OPTIONS)
  endif()
  string(TOUPPER "${CMAKE_BUILD_TYPE}" T39_BUILD_TYPE)
  set(T39_ABI_MATERIAL "key=1\nsource=${T39_SOURCE_ID}\ntarget=${T39_TARGET}\nmath=${T39_MATH_OPTIONS}\ntarget_options=${T39_OPTIONS}\ntarget_definitions=${T39_DEFS}\nbody_definitions=${T39_BODY_DEFS}\nbody_options=${T39_BODY_OPTIONS}\nhip_definitions=${T39_HIP_DEFS}\nhip_options=${T39_HIP_OPTIONS}\narch=${CMAKE_HIP_ARCHITECTURES}\nconfig=${CMAKE_BUILD_TYPE}\ncxx_flags=${CMAKE_CXX_FLAGS}\ncxx_config_flags=${CMAKE_CXX_FLAGS_${T39_BUILD_TYPE}}\nhip_flags=${CMAKE_HIP_FLAGS}\nhip_config_flags=${CMAKE_HIP_FLAGS_${T39_BUILD_TYPE}}\ncxx_compiler=${CMAKE_CXX_COMPILER_ID}|${CMAKE_CXX_COMPILER_VERSION}\nhip_compiler=${CMAKE_HIP_COMPILER_ID}|${CMAKE_HIP_COMPILER_VERSION}\ngraph_trace=undefined\n")
  string(SHA256 T39_ABI "${T39_ABI_MATERIAL}")
  file(WRITE "${CMAKE_BINARY_DIR}/${T39_TARGET}-graph-abi-material.txt" "${T39_ABI_MATERIAL}")
  file(WRITE "${CMAKE_BINARY_DIR}/${T39_TARGET}-graph-abi.txt" "${T39_ABI}\n")
  target_compile_definitions(${T39_TARGET} PRIVATE STRATA_GRAPH_BUILD_ABI="${T39_ABI}")
 endif()
endforeach()
