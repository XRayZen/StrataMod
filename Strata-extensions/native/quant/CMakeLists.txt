cmake_minimum_required(VERSION 3.24)
project(strata_t09 LANGUAGES CXX HIP)
find_package(hip CONFIG REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MATERIAL "")
foreach(INPUT Strata-extensions/native/quant/quant.hpp Strata-extensions/native/quant/quant.hip Strata-extensions/tests/quant/native_fixture.cpp)
  file(SHA256 "${ROOT}/${INPUT}" HASH)
  string(APPEND MATERIAL "${INPUT}|${HASH}\n")
endforeach()
string(SHA256 SOURCE_ID "${MATERIAL}")
add_executable(t09-quant quant.hip ../../tests/quant/native_fixture.cpp ../artifact/native_reader.cpp)
target_include_directories(t09-quant PRIVATE . ../artifact "${ROOT}/include")
target_compile_features(t09-quant PRIVATE cxx_std_17)
set_target_properties(t09-quant PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_options(t09-quant PRIVATE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
target_compile_definitions(t09-quant PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T09_SOURCE_ID="${SOURCE_ID}")
target_link_libraries(t09-quant PRIVATE hip::host psapi)
