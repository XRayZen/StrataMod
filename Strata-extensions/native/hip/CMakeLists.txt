cmake_minimum_required(VERSION 3.24)
project(strata_ext_hip_fixture LANGUAGES CXX HIP)
if(NOT WIN32)
  message(FATAL_ERROR "T07 fixture scope is Windows HIP")
endif()
find_package(hip CONFIG REQUIRED)
find_package(hipblas CONFIG REQUIRED)
get_filename_component(WORKTREE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(INPUTS Strata-extensions/native/hip/runtime.hpp Strata-extensions/tests/hip/backend_fixture.hip)
set(MATERIAL "")
foreach(INPUT IN LISTS INPUTS)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WORKTREE_ROOT}/${INPUT}")
  file(SHA256 "${WORKTREE_ROOT}/${INPUT}" INPUT_HASH)
  string(APPEND MATERIAL "${INPUT}|${INPUT_HASH}\n")
endforeach()
string(SHA256 SOURCE_ID "${MATERIAL}")
add_executable(strata-ext-hip-fixture ../../tests/hip/backend_fixture.hip)
target_include_directories(strata-ext-hip-fixture PRIVATE .)
target_compile_features(strata-ext-hip-fixture PRIVATE cxx_std_17)
set_target_properties(strata-ext-hip-fixture PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
target_compile_definitions(strata-ext-hip-fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
  STRATA_EXT_SOURCE_ID="${SOURCE_ID}")
target_link_libraries(strata-ext-hip-fixture PRIVATE hip::host roc::hipblas dxgi)
