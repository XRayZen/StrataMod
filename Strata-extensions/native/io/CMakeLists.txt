cmake_minimum_required(VERSION 3.20)
project(t17_span LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MATERIAL "")
foreach(INPUT Strata-extensions/native/io/span_provider.hpp Strata-extensions/native/io/span_provider.cpp Strata-extensions/tests/io/span_fixture.cpp)
  file(SHA256 "${ROOT}/${INPUT}" HASH)
  string(APPEND MATERIAL "${INPUT}|${HASH}\n")
endforeach()
string(SHA256 SOURCE_ID "${MATERIAL}")
add_executable(t17-span span_provider.cpp ../../tests/io/span_fixture.cpp ../artifact/native_reader.cpp ../cache/lease_queue.cpp ../stats/measurement.cpp "${ROOT}/src/platform/direct_file.cpp")
target_include_directories(t17-span PRIVATE . ../artifact ../cache ../stats "${ROOT}/include")
target_compile_features(t17-span PRIVATE cxx_std_17)
target_compile_options(t17-span PRIVATE /W4 /EHsc /fp:strict)
set_property(TARGET t17-span PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_definitions(t17-span PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN T17_SOURCE_ID="${SOURCE_ID}")
set_target_properties(t17-span PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bundle")
